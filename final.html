My Console-Based Java Game

Tic-Tac-Toe is a two-player strategy game played on a 3×3 board. In this Java console version, the player uses X, while the computer uses O. The goal is to get three matching symbols in a horizontal, vertical, or diagonal line. Players take turns choosing an empty position from 1 to 9. The game ends when either the player or computer wins, or when all spaces are filled and the result will show a draw.


import java.util.Scanner;
import java.util.Random;

public class LuckyNumberGame {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        Random rand = new Random();

        int luckyNumber = rand.nextInt(10) + 1;
        int score = 0;

        System.out.println("=======================");
        System.out.println("  LUCKY NUMBER GAME");
        System.out.println("~~~~~~~~~~~~~~~~~~~~~~~");
        System.out.println("Guess the lucky number!");
        System.out.println("Choose a number from 1 to 10.");
        System.out.println("You have 3 chances.");

        for (int i = 1; i <= 3; i++) {

            System.out.print("\nAttempt " + i + ": ");
            int guess = sc.nextInt();

            if (guess == luckyNumber) {
                System.out.println("Amazing! You just found it!");
                score = 10;
                break;
            } else if (guess < luckyNumber) {
                System.out.println("Too low! Try higher.");
            } else {
                System.out.println("Too high! Try lower.");
            }
        }

        System.out.println();
        System.out.println("       GAME RESULTS");
        System.out.println();
        System.out.println("Lucky Number: " + luckyNumber);
        System.out.println("Your Score: " + score);

        if (score == 10) {
            System.out.println("YOU ARE LUCKY!");
        } else {
            System.out.println("Better luck next time!");
        }

        
        System.out.println("   THANK YOU FOR PLAYING!");
        sc.close();
    }
}
